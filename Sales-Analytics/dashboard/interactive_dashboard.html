Interactive SQL Query Sandbox & Verification
Calculates annual revenue, profit margin, units, and year-over-year percentage growth rates. 
WITH YearlyMetrics AS (
    SELECT 
        Year,
        ROUND(SUM(Sales), 2) AS Total_Sales,
        ROUND(SUM(Profit), 2) AS Total_Profit,
        ROUND(SUM(Profit) * 100.0 / SUM(Sales), 2) AS Profit_Margin_Pct,
        SUM(Quantity) AS Total_Units
    FROM sales_orders
    GROUP BY Year
)
SELECT 
    Year,
    Total_Sales,
    LAG(Total_Sales) OVER (ORDER BY Year) AS Prior_Year_Sales,
    ROUND((Total_Sales - LAG(Total_Sales) OVER (ORDER BY Year)) * 100.0 / LAG(Total_Sales) OVER (ORDER BY Year), 2) AS Sales_YoY_Growth_Pct,
    Profit_Margin_Pct,
    Total_Units
FROM YearlyMetrics;
